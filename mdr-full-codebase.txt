import React, { useState, useEffect } from 'react';
import { View, ActivityIndicator } from 'react-native';
import { Session } from '@supabase/supabase-js';
import * as Linking from 'expo-linking';
import { supabase } from './lib/supabase';

import LoginScreen from './screens/LoginScreen';
import SignUpScreen from './screens/SignUpScreen';
import UpdatePasswordScreen from './screens/UpdatePasswordScreen';
import MainLayout from './screens/MainLayout';

export default function App() {
  const [session, setSession] = useState<Session | null>(null);
  const [loading, setLoading] = useState(true);
  const [authScreen, setAuthScreen] = useState<'Login' | 'SignUp' | 'UpdatePassword'>('Login');

  useEffect(() => {
    // 1. Exchange deep-link codes or apply tokens from implicit-flow links.
    const handleUrl = async (event: { url: string }) => {
      if (event.url) {
        const url = new URL(event.url);
        const code = url.searchParams.get('code');

        if (code) {
          await supabase.auth.exchangeCodeForSession(code);
          return;
        }

        const params = new URLSearchParams(url.hash.slice(1));
        const access_token = params.get('access_token');
        const refresh_token = params.get('refresh_token');
        if (access_token && refresh_token) {
          await supabase.auth.setSession({ access_token, refresh_token });
        }
      }
    };

    // Catch the URL if the app was completely closed
    Linking.getInitialURL().then((url) => {
      if (url) handleUrl({ url });
    });
    
    // Catch the URL if the app was already open in the background
    const urlSubscription = Linking.addEventListener('url', handleUrl);

    // 2. Standard session setup
    supabase.auth.getSession().then(({ data: { session } }) => {
      setSession(session);
      setLoading(false);
    });

    // 3. Listen for the 'PASSWORD_RECOVERY' event triggered by the URL
    const { data: authSubscription } = supabase.auth.onAuthStateChange((event, session) => {
      setSession(session);
      if (event === 'PASSWORD_RECOVERY') {
        setAuthScreen('UpdatePassword');
      } else {
        setAuthScreen('Login');
      }
    });

    return () => {
      urlSubscription.remove();
      authSubscription.subscription.unsubscribe();
    };
  }, []);

  if (loading) return <View style={{ flex: 1, justifyContent: 'center', alignItems: 'center', backgroundColor: '#f4effc' }}><ActivityIndicator size="large" color="#5138a1" /></View>;

  if (!session || authScreen === 'UpdatePassword') {
    if (authScreen === 'UpdatePassword') return <UpdatePasswordScreen onComplete={() => setAuthScreen('Login')} />;
    if (authScreen === 'SignUp') return <SignUpScreen onBack={() => setAuthScreen('Login')} />;
    return <LoginScreen onNavigateToSignUp={() => setAuthScreen('SignUp')} />;
  }

  return <MainLayout />;
}

const smoothLayoutAnim = {
  duration: 250,
  create: { type: LayoutAnimation.Types.easeInEaseOut, property: LayoutAnimation.Properties.opacity },
  update: { type: LayoutAnimation.Types.easeInEaseOut },
  delete: { type: LayoutAnimation.Types.easeInEaseOut, property: LayoutAnimation.Properties.opacity },
};

export default function CatalogScreen({ activeCategory, openQuantityModal }: any) {
  const [items, setItems] = useState<any[]>([]);
  const [filteredItems, setFilteredItems] = useState<any[]>([]);
  const [allSubcategories, setAllSubcategories] = useState<any[]>([]);
  const [displaySubcategories, setDisplaySubcategories] = useState<any[]>([]);
  
  const [activeSubcategory, setActiveSubcategory] = useState<number | null>(null);
  const [searchQuery, setSearchQuery] = useState('');
  const [isLoading, setIsLoading] = useState(true);

  useEffect(() => {
    fetchCatalogData();
  }, []);

  useEffect(() => {
    let result = items;
    if (activeCategory !== null) result = result.filter(item => item.category_id === activeCategory.id);
    if (activeSubcategory !== null) result = result.filter(item => item.subcategory_id === activeSubcategory);
    if (searchQuery.trim() !== '') {
      result = result.filter(item => {
        const query = searchQuery.toLowerCase();
        return (item.name.toLowerCase().includes(query) || (item.item_code && item.item_code.toLowerCase().includes(query)) || (item.search_keywords && item.search_keywords.toLowerCase().includes(query)));
      });
    }
    LayoutAnimation.configureNext(smoothLayoutAnim);
    setFilteredItems(result);
  }, [searchQuery, activeCategory, activeSubcategory, items]);

  useEffect(() => {
    LayoutAnimation.configureNext(smoothLayoutAnim);
    if (activeCategory === null) {
      setDisplaySubcategories([]);
    } else {
      setDisplaySubcategories(allSubcategories.filter(sub => sub.category_id === activeCategory.id));
    }
    setActiveSubcategory(null);
  }, [activeCategory, allSubcategories]);

  const fetchCatalogData = async () => {
    setIsLoading(true);
    const { data: subData } = await supabase.from('subcategories').select('id, category_id, name').order('sort_order', { ascending: true });
    if (subData) setAllSubcategories(subData);

    const { data: itemData } = await supabase.from('items').select('id, item_code, name, current_stock, category_id, subcategory_id, unit, search_keywords').eq('is_active', true).order('name');
    if (itemData) { setItems(itemData); setFilteredItems(itemData); }
    setIsLoading(false);
  };

  const renderItem = ({ item }: { item: any }) => (
    <View style={styles.card}>
      <View style={styles.cardInfo}>
        <Text style={styles.itemCode}>{item.item_code}</Text>
        <Text style={styles.itemName}>{item.name}</Text>
        <Text style={styles.itemStock}>Baki: {item.current_stock} {item.unit}</Text>
      </View>
      <TouchableOpacity activeOpacity={0.7} style={styles.button} onPress={() => openQuantityModal(item)}>
        <Text style={styles.buttonText}>Pilih</Text>
      </TouchableOpacity>
    </View>
  );

  return (
    <View style={styles.container}>
      <View style={styles.searchContainer}>
        <View style={styles.searchWrapper}>
          <Feather name="search" size={20} color="#9ca3af" style={styles.searchIcon} />
          <TextInput style={styles.searchInput} placeholder="Cari item atau kod..." placeholderTextColor="#9ca3af" value={searchQuery} onChangeText={setSearchQuery} />
        </View>
      </View>

      {displaySubcategories.length > 0 ? (
        <View style={styles.subcategoryContainer}>
          <ScrollView horizontal showsHorizontalScrollIndicator={false} contentContainerStyle={styles.categoryScroll}>
            <TouchableOpacity activeOpacity={0.7} style={[styles.categoryPill, activeSubcategory === null && styles.activePill]} onPress={() => { LayoutAnimation.configureNext(smoothLayoutAnim); setActiveSubcategory(null); }}>
              <Text style={[styles.categoryText, activeSubcategory === null && styles.activeText]}>Semua</Text>
            </TouchableOpacity>
            {displaySubcategories.map((sub) => (
              <TouchableOpacity key={sub.id} activeOpacity={0.7} style={[styles.categoryPill, activeSubcategory === sub.id && styles.activePill]} onPress={() => { LayoutAnimation.configureNext(smoothLayoutAnim); setActiveSubcategory(sub.id); }}>
                <Text style={[styles.categoryText, activeSubcategory === sub.id && styles.activeText]}>{sub.name}</Text>
              </TouchableOpacity>
            ))}
          </ScrollView>
        </View>
      ) : null}
      
      {isLoading ? (
        <ActivityIndicator size="large" color="#5138a1" style={{ marginTop: 50 }} />
      ) : (
        <FlatList data={filteredItems} keyExtractor={(item) => item.id.toString()} renderItem={renderItem} contentContainerStyle={styles.list} ListEmptyComponent={<Text style={styles.emptyText}>Tiada item dijumpai.</Text>} />
      )}
    </View>
  );
}

    // 2. Add safety check here too

      setLoading(false);
      return;

    const { data: reqs, error } = await supabase
      .from('requests')
      .select(`
        id, 
        status, 
        request_date, 
        request_items ( *, items (*) )
      `)
      .eq('user_id', userProfile.id)
      .order('request_date', { ascending: false });

export default function SignUpScreen({ onBack }: { onBack: () => void }) {
  const [name, setName] = useState('');
  const [jawatan, setJawatan] = useState('');
  const [deptId, setDeptId] = useState<number | null>(null);
  const [email, setEmail] = useState('');
  const [password, setPassword] = useState('');
  const [confirmPassword, setConfirmPassword] = useState('');
  
  const [showPassword, setShowPassword] = useState(false);
  const [loading, setLoading] = useState(false);
  const [showDeptModal, setShowDeptModal] = useState(false);
  
  // New visual error state
  const [errorMsg, setErrorMsg] = useState('');

  async function handleSignUp() {
    setErrorMsg(''); // Clear old errors
    
    if (!name || !jawatan || !deptId || !email || !password || !confirmPassword) {
      return setErrorMsg('Sila isi semua maklumat.');
    }
    if (password !== confirmPassword) {
      return setErrorMsg('Kata laluan tidak sepadan.');
    }

    setLoading(true);
    
    // 1. Create Auth Account
    const { data: authData, error: authError } = await supabase.auth.signUp({ email, password });
    
    if (authError) {
      setLoading(false);
      return setErrorMsg(authError.message);
    }

    if (authData.user) {
      // 2. Insert into public.users table (Auto-forced to 'User' role)
      const { error: dbError } = await supabase.from('users').insert([{
        id: authData.user.id,
        full_name: name,
        jawatan: jawatan,
        department_id: deptId,
        role: 'User', // <-- Forces every new account to be a normal user
        avatar: 'default'
      }]);

      if (dbError) {
        setErrorMsg('Ralat Pangkalan Data: ' + dbError.message);
      } else {
        alert('Pendaftaran Berjaya! Sila log masuk.');
        onBack();
      }
    }
    setLoading(false);
  }

export async function loginOfficer(formData: FormData) {
  const email = formData.get('email') as string;
  const password = formData.get('password') as string;
  
  // Initialize the Supabase server client
  const supabase = await createClient();

  const { data: authData, error: authError } = await supabase.auth.signInWithPassword({
    email,
    password,
  });

  if (authError) {
    return { error: 'E-mel atau kata laluan tidak sah.' };
  }

  // Verify the user is an Officer
  if (authData.user) {
    const { data: profile } = await supabase
      .from('users')
      .select('role') // We will create this column next
      .eq('id', authData.user.id)
      .single();

    if (profile?.role === 'Officer' || profile?.role === 'admin') {
      redirect('/dashboard');
    } else {
      // Kick them out if they are just normal mobile staff
      await supabase.auth.signOut();
      return { error: 'Akses ditolak. Portal ini hanya untuk Pegawai Pelulus MDR.' };
    }
  }
  
  return { error: 'Ralat pelayan.' };
}

'use server';

import { revalidatePath } from 'next/cache';
import { createClient } from '../../../../lib/supabase/server';

type Line = { id: number; approved_qty: number };
export type ProcessResult = { error: string } | { ok: true };

export async function processRequest(
  requestId: number,
  decision: 'Approved' | 'Rejected',
  lines: Line[] = [],
  reason = ''
): Promise<ProcessResult> {
  if (!Number.isInteger(requestId) || requestId <= 0) return { error: 'ID permohonan tidak sah.' };

  const supabase = await createClient();
  const { data: { user } } = await supabase.auth.getUser();
  if (!user) return { error: 'Sesi tamat. Sila log masuk semula.' };

  if (decision === 'Approved') {
    const valid = lines.every((l) => Number.isInteger(l.id) && Number.isInteger(l.approved_qty) && l.approved_qty >= 0);
    if (!valid) return { error: 'Kuantiti tidak sah.' };

    const { error } = await supabase.rpc('approve_request_lines', {
      p_request_id: requestId,
      p_lines: lines.map((l) => ({ id: l.id, qty: l.approved_qty })),
    });
    if (error) return { error: error.message };
  } else {
    if (!reason.trim()) return { error: 'Sila nyatakan sebab penolakan.' };

    const { error } = await supabase.rpc('reject_request', {
      p_request_id: requestId,
      p_reason: reason.trim(),
    });
    if (error) return { error: error.message };
  }

  revalidatePath('/dashboard');
  revalidatePath(`/dashboard/requests/${requestId}`);
  return { ok: true }; // the caller decides where to go next
}

  const parts = new Intl.DateTimeFormat('en-GB', {

'use client';

import { useEffect, useState } from 'react';

const TZ = 'Asia/Kuala_Lumpur';
const MONTHS = ['Januari', 'Februari', 'Mac', 'April', 'Mei', 'Jun', 'Julai', 'Ogos', 'September', 'Oktober', 'November', 'Disember'];
const DAYS = ['Isn', 'Sel', 'Rab', 'Kha', 'Jum', 'Sab', 'Ahd'];

function kualaLumpurToday() {
  const p = new Intl.DateTimeFormat('en-CA', {
    timeZone: TZ, year: 'numeric', month: '2-digit', day: '2-digit',
  }).formatToParts(new Date());
  const get = (t: string) => Number(p.find((x) => x.type === t)?.value);
  return { y: get('year'), m: get('month') - 1, d: get('day') };
}

const pad = (n: number) => String(n).padStart(2, '0');

/** markedDates: array of 'YYYY-MM-DD' (e.g. days that have pending requests) */
export default function MiniCalendar({ markedDates = [] }: { markedDates?: string[] }) {
  const [today, setToday] = useState<{ y: number; m: number; d: number } | null>(null);
  const [view, setView] = useState<{ y: number; m: number } | null>(null);

  useEffect(() => {
    const t = kualaLumpurToday();
    setToday(t);
    setView({ y: t.y, m: t.m });
  }, []);

  const shell =
    'rounded-3xl border border-white/80 bg-white/70 p-5 shadow-[0_8px_24px_-8px_rgba(40,70,110,0.18)]';

  if (!today || !view) return <div className={`${shell} h-[360px] animate-pulse`} />;

  const first = new Date(view.y, view.m, 1);
  const offset = (first.getDay() + 6) % 7; // Monday first
  const daysInMonth = new Date(view.y, view.m + 1, 0).getDate();
  const daysInPrev = new Date(view.y, view.m, 0).getDate();

  const cells: { day: number; current: boolean; key: string }[] = [];
  for (let i = offset - 1; i >= 0; i--) cells.push({ day: daysInPrev - i, current: false, key: `p${i}` });
  for (let d = 1; d <= daysInMonth; d++) cells.push({ day: d, current: true, key: `c${d}` });
  while (cells.length % 7 !== 0) cells.push({ day: cells.length % 7 + 1, current: false, key: `n${cells.length}` });

  const move = (delta: number) => {
    const d = new Date(view.y, view.m + delta, 1);
    setView({ y: d.getFullYear(), m: d.getMonth() });
  };

  return (
    <div className={shell}>
      <div className="mb-3 flex items-center justify-between">
        <p className="text-base font-medium text-slate-600">
          {MONTHS[view.m]} {view.y}
        </p>
        <div className="flex items-center gap-1">
          <button onClick={() => move(-1)} aria-label="Bulan sebelum"
            className="rounded-lg p-1.5 text-slate-500 hover:bg-white/80">
            <svg className="h-4 w-4" fill="none" viewBox="0 0 24 24" stroke="currentColor" strokeWidth="1.8"><path strokeLinecap="round" strokeLinejoin="round" d="M15 6l-6 6 6 6" /></svg>
          </button>
          <button onClick={() => move(1)} aria-label="Bulan seterusnya"
            className="rounded-lg p-1.5 text-slate-500 hover:bg-white/80">
            <svg className="h-4 w-4" fill="none" viewBox="0 0 24 24" stroke="currentColor" strokeWidth="1.8"><path strokeLinecap="round" strokeLinejoin="round" d="M9 6l6 6-6 6" /></svg>
          </button>
        </div>
      </div>

      <div className="grid grid-cols-7 gap-y-1 text-center">
        {DAYS.map((d) => (
          <span key={d} className="pb-2 text-xs font-medium text-slate-400">{d}</span>
        ))}
        {cells.map((c) => {
          const iso = `${view.y}-${pad(view.m + 1)}-${pad(c.day)}`;
          const isToday = c.current && view.y === today.y && view.m === today.m && c.day === today.d;
          const marked = c.current && markedDates.includes(iso);
          return (
            <div key={c.key} className="flex justify-center">
              <span
                className={`relative flex h-9 w-9 items-center justify-center rounded-xl text-sm ${
                  isToday
                    ? 'bg-[#111936] font-semibold text-white shadow-md'
                    : c.current
                    ? 'text-slate-700 hover:bg-white/80'
                    : 'text-slate-300'
                }`}
              >
                {c.day}
                {marked && (
                  <span className={`absolute bottom-1 h-1 w-1 rounded-full ${isToday ? 'bg-white' : 'bg-amber-500'}`} />
                )}
              </span>
            </div>
          );
        })}
      </div>

      <button
        onClick={() => setView({ y: today.y, m: today.m })}
        className="mt-3 w-full rounded-xl py-1.5 text-xs font-medium text-slate-500 hover:bg-white/70"
      >
        Hari ini
      </button>
    </div>
  );
}

              cookieStore.set(name, value, options)
            )

            // The `setAll` method was called from a Server Component.
            // This can be ignored if you have middleware refreshing
            // user sessions.
          }
        },
      },
    }
  )
}
